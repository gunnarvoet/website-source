{{/* ------------------------------------------------------------------ Returns the pages related to the given page, as a dict of slices keyed by section: {posts, publications, talks}. Two real links exist in the content and both are used. Posts name their project outright in `projects`, matching the bundle folder name; and projects, publications and posts all share `tags`. The union of the two is deduplicated by Hugo's `union`, then the page itself is dropped. ------------------------------------------------------------------ */}} {{- $page := . -}} {{- $found := slice -}} {{- with $page.Params.tags -}} {{- $found = $found | union (where site.RegularPages "Params.tags" "intersect" .) -}} {{- end -}} {{- with $page.File -}} {{- $found = $found | union (where site.RegularPages "Params.projects" "intersect" (slice .ContentBaseName)) -}} {{- end -}} {{- $found = where $found "Permalink" "!=" $page.Permalink -}} {{- return dict "posts" (sort (where $found "Section" "post") "Date" "desc") "publications" (sort (where $found "Section" "publication") "Date" "desc") "talks" (sort (where $found "Section" "talk") "Date" "desc") -}}
